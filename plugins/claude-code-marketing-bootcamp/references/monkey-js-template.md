# monkey.js — File Format and Template

This is the format for `monkey.js` files written by the Sniffer skill and appended to by the Replay skill.

## Core Principles

- **Proof-based, not speculative** — every function in this file has returned HTTP 2xx at least once in a real session
- **Ambient auth only** — all fetch calls use `credentials: "include"`, never hardcoded tokens or API keys
- **Self-contained** — each function can be copy-pasted into the Claude-in-Chrome `javascript_tool` and run standalone
- **Documented** — JSDoc comments describe what each function does, its parameters, and expected return shape

## File Template

```javascript
/**
 * monkey.js — [SITE NAME]
 * Generated by Sniffer and Monkey
 * Sniffed: [DATE]
 * Last proven run: [DATE]
 *
 * Usage: These functions run inside the Claude-in-Chrome javascript_tool on a tab at [SITE URL].
 * Auth is ambient — the browser session must be authenticated.
 * Never add API keys or tokens here.
 */

const ORIGIN = location.origin; // e.g., "https://example.com"

// ─────────────────────────────────────────────────────────────────────────────
// BASE HELPER — always present
// ─────────────────────────────────────────────────────────────────────────────

/**
 * Low-level API caller. All other functions use this.
 * @param {string} method - HTTP method (GET, POST, PUT, DELETE, PATCH)
 * @param {string} path - URL path, e.g. "/api/v1/posts" or "/api/v1/drafts/189289551"
 * @param {object|null} body - JSON body (or null for GET/DELETE)
 * @returns {{ status: number, data: any }}
 */
async function callAPI(method, path, body = null) {
  const options = {
    method,
    headers: { "Content-Type": "application/json" },
    credentials: "include"
  };
  if (body !== null) {
    options.body = JSON.stringify(body);
  }
  const res = await fetch(ORIGIN + path, options);
  let data;
  try { data = await res.json(); } catch { data = null; }
  return { status: res.status, data };
}


// ─────────────────────────────────────────────────────────────────────────────
// PROVEN FUNCTIONS — added after confirmed successful run
// ─────────────────────────────────────────────────────────────────────────────

// Each function below was proven working in a real session.
// Format: JSDoc → async function → proven return shape comment

/**
 * [FUNCTION NAME] — [one-line description]
 * Endpoint: [METHOD] [NORMALIZED PATH]
 * Proven: [DATE] — HTTP [STATUS]
 *
 * @param {string} paramName - description
 * @returns {{ status: number, data: { field: type, ... } }}
 */
async function exampleFunction(paramName) {
  return await callAPI("GET", `/api/v1/resource/${paramName}`);
  // Returns: { status: 200, data: { id: number, name: string, ... } }
}
```

## Real Example — Substack Publisher

This is a real `monkey.js` built from the Substack publisher API session:

```javascript
/**
 * monkey.js — Substack (heymitch.substack.com)
 * Generated by Sniffer and Monkey
 * Sniffed: 2026-02-26
 * Last proven run: 2026-02-26
 *
 * Usage: Run inside the Claude-in-Chrome javascript_tool on any tab at heymitch.substack.com
 * Auth is ambient — must be logged into Substack.
 */

const ORIGIN = location.origin; // "https://heymitch.substack.com"

async function callAPI(method, path, body = null) {
  const options = {
    method,
    headers: { "Content-Type": "application/json" },
    credentials: "include"
  };
  if (body !== null) {
    options.body = JSON.stringify(body);
  }
  const res = await fetch(ORIGIN + path, options);
  let data;
  try { data = await res.json(); } catch { data = null; }
  return { status: res.status, data };
}

/**
 * updateDraft — Update an existing draft's title, subtitle, and body
 * Endpoint: PUT /api/v1/drafts/:id
 * Proven: 2026-02-26 — HTTP 200
 *
 * @param {number} draftId - The numeric draft ID (visible in the URL when editing)
 * @param {string} title - Draft title
 * @param {string} subtitle - Draft subtitle/description
 * @param {string} bodyText - Plain text body (auto-wrapped in ProseMirror format)
 * @returns {{ status: number, data: { id: number, draft_title: string, draft_subtitle: string } }}
 */
async function updateDraft(draftId, title, subtitle, bodyText) {
  const prosemirrorBody = JSON.stringify({
    type: "doc",
    content: [
      {
        type: "paragraph",
        content: [{ type: "text", text: bodyText }]
      }
    ]
  });
  return await callAPI("PUT", `/api/v1/drafts/${draftId}`, {
    draft_title: title,
    draft_subtitle: subtitle,
    draft_podcast_url: null,
    draft_podcast_duration: null,
    draft_body: prosemirrorBody
  });
  // Returns: { status: 200, data: { id: 189289551, draft_title: "...", ... } }
}

/**
 * getDrafts — List all drafts for this publication
 * Endpoint: GET /api/v1/drafts
 * Proven: 2026-02-26 — HTTP 200
 *
 * @returns {{ status: number, data: Array<{ id: number, draft_title: string, draft_created_at: string }> }}
 */
async function getDrafts() {
  return await callAPI("GET", "/api/v1/drafts");
  // Returns: { status: 200, data: [ { id: ..., draft_title: "...", ... }, ... ] }
}
```

## Notes for Adding New Functions

When the Replay skill proves a new call works, add a function following this checklist:

- [ ] JSDoc with `@param` for every parameter and `@returns` with actual shape
- [ ] "Proven: [DATE] — HTTP [STATUS]" in the JSDoc
- [ ] Uses `callAPI()` helper — no raw `fetch()` calls directly
- [ ] No hardcoded IDs, tokens, or user-specific data as defaults
- [ ] Comment at the end showing real return shape from the proven run
